1    from flask import Flask, render_template  # Flask是一个web框架，为用户提供工具、库和技术来构建一个web应用程序。
2    from flask_script import Server, Manager  # flask_script可以通过命令行的形式来操作Flask
3    from flask_bootstrap import Bootstrap  # flask_bootstrap提供的用户界面组件可用于创建整洁且具有吸引力的网页
4    # Bootstrap是一个集成了HTML标记、CSS样式及JavaScript行为的前端开发框架，无需反复写模板、样式等
5    from flask_moment import Moment  # flask-Moment是一个flask的扩展模块，用来处理时间日期等信息。
6    # flask-wtf是集成wtforms，并带有csrf令牌的安全表单和全局的csrf保护的功能。
7    # 在建立表单所创建的类都是继承于flask_wtf中的FlaskForm，而FlaskForm是继承wtforms中forms。
8    from flask_wtf import FlaskForm
9    from wtforms import StringField, SubmitField
10   from wtforms.validators import DataRequired
11   
12   import sys
13   sys.path.insert(0, "../")
14   # sys.path.insert(0, path)将path添加到检索范围的特定位置（0即第一个检索），否则默认从python安装文件夹开始检索
15   # 以下import aiml会从当前文件夹中导入aiml模块，而不是python库安装的位置
16   import aiml  # aiml人工智能标记语言
17   
18   k = aiml.Kernel()  # 创建机器人k
19   k.learn("cn-startup.xml") # 当前文件夹中有个启动文件cn-startup.xml，作为加载AIML文件的主入口点
20   k.respond("load aiml cn") # k对指令做出回应
21   k.respond("start")        # k对指令做出回应
22                            
23   app = Flask(__name__)  # 创建Flask实例，__name__是系统变量，该变量指的是本py文件的文件名
24   app.config['SECRET_KEY'] = 'hard to guess string'
25   # 为了实现CSRF保护，Flask-WTF需要程序设置一个密钥。Flask-WTF使用这个密钥生成加密令牌，再用令牌验证请求中表单数据的真伪。
26   
27   manager = Manager(app)  # 创建Manager管理类的对象
28   server = Server(host="127.0.0.1", port=80, threaded=True)
29   # 只能在本机通过127.0.0.1:port来访问。如果写0.0.0.0，就可以用127.0.0.1:port、内网:port、外网:port来访问flask接口。
30   manager.add_command("runserver", server)  # 可用命令行 "python 本文件名.py runserver" 来运行程序
31   bootstrap = Bootstrap(app)  # 创建Bootstrap对象的对象
32   moment = Moment(app)  # 创建Moment对象的实例
33   
34   
35   class NameForm(FlaskForm):   # 定义一个表单类NameForm（继承于FlaskForm）
36       name = StringField('请开始交谈：', validators=[DataRequired()]) # 定义了一个文本输入框，validators=..表单提交时必须非空
37       submit = SubmitField('提交')  # 定义了一个按钮，显示“提交”
38   
39   @app.route('/', methods=['GET', 'POST'])
40   def index():
41       name = ''
42       form = NameForm()  # 用NameForm类创建了一个表单实例form
43       if form.validate_on_submit(): # 有了form.validate_on_submit()方法，就无需通过判断请求方法是否为”POST”来判断表单是否提交
44           name = form.name.data #form.name就是表单中的文本输入框
45           form.name.data = ''  #清空文本输入框的内容
46       return render_template('index.html', form=form, name=k.respond(name)) # k.respond(name)是对输入内容的回答
47   
48   if __name__ == '__main__':
49       manager.run()   # 用命令行 "python 本文件名.py runserver" 来运行程序
